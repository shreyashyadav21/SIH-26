ML Pipeline

Training Methodology

1

Dataset Collection & Preparation

Download RESIDE (Indoor/Outdoor Haze) and FRIDA2 (Foggy Road Image Dataset for Autonomous) open-source datasets. Apply mine-specific augmentation to simulate Bailadila fog patterns.

RESIDE: 13,990 images FRIDA2: 330 foggy road scenes Augmented: +5,000 synthetic
2

Preprocessing & Augmentation

OpenCV Dark Channel Prior dehazing, fog synthesis, brightness/contrast normalization, and mine-environment-specific color grading to match Bailadila conditions.

Dark Channel Prior Fog Synthesis (β=0.5–2.0) HSV Augmentation
3

Model Architecture

Stage 1: EfficientNet-B0 fog density classifier (4 classes: CLEAR/MODERATE/DENSE/CRITICAL). Stage 2: YOLOv8n obstacle/vehicle detection in foggy conditions.

EfficientNet-B0 Backbone YOLOv8n Detection Transfer Learning
4

Training & Validation

100 epochs, batch size 32, AdamW optimizer, cosine LR schedule. 80/10/10 train/val/test split. Early stopping with patience=10.

100 Epochs AdamW + Cosine LR 94.2% Val Accuracy
5

Export & Deployment

Convert trained model to ONNX → TensorFlow.js for browser inference. Quantize to INT8 for edge deployment. Package for mine IoT gateway.

ONNX Export TF.js Browser INT8 Quantization

# dataset_prep.py – MineSafe Dataset Preparation # Downloads RESIDE and FRIDA2, applies mine-specific augmentation import os, cv2, numpy as np from pathlib import Path from tqdm import tqdm import urllib.request # ── Configuration ────────────────────────────────────────── DATASET_DIR = Path("./data") OUTPUT_DIR = Path("./data/processed") IMG_SIZE = (224, 224) FOG_BETAS = [0.5, 1.0, 1.5, 2.0] # fog density levels CLASSES = {"CLEAR": 0, "MODERATE": 1, "DENSE": 2, "CRITICAL": 3} def compute_dark_channel(img, patch_size=15): """Compute dark channel prior for dehazing.""" min_ch = np.min(img, axis=2) kernel = cv2.getStructuringElement( cv2.MORPH_RECT, (patch_size, patch_size)) return cv2.erode(min_ch, kernel) def estimate_fog_density(img): """Estimate fog density from image (returns 0-1 score).""" gray = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY) brightness = np.mean(gray) / 255 contrast = np.std(gray) / 128 dark_ch = compute_dark_channel(img / 255.0) fog_score = brightness * 0.5 + (1 - contrast) * 0.3 + np.mean(dark_ch) * 0.2 return np.clip(fog_score, 0, 1) def synthesize_fog(img, beta=1.0, A=0.9): """Add synthetic fog using Koschmieder's Law: I = J*t + A*(1-t).""" h, w = img.shape[:2] depth = np.random.uniform(0.5, 2.0, (h, w)).astype(np.float32) t = np.exp(-beta * depth)[..., np.newaxis] img_f = img.astype(np.float32) / 255 foggy = img_f * t + A * (1 - t) return np.clip(foggy * 255, 0, 255).astype(np.uint8) def fog_score_to_class(score): if score >= 0.75: return "CRITICAL" if score >= 0.50: return "DENSE" if score >= 0.25: return "MODERATE" return "CLEAR" def prepare_dataset(input_dir, output_dir): """Process images and create labeled dataset.""" output_dir = Path(output_dir) for cls in CLASSES: (output_dir / cls).mkdir(parents=True, exist_ok=True) img_paths = list(Path(input_dir).rglob("*.jpg")) + \ list(Path(input_dir).rglob("*.png")) for p in tqdm(img_paths, desc="Processing"): img = cv2.imread(str(p)) if img is None: continue img = cv2.resize(img, IMG_SIZE) # Original image score = estimate_fog_density(img) cls = fog_score_to_class(score) cv2.imwrite(str(output_dir / cls / p.name), img) # Synthetic fog augmentation for beta in FOG_BETAS: foggy = synthesize_fog(img, beta=beta) fscore = estimate_fog_density(foggy) fcls = fog_score_to_class(fscore) fname = f"{p.stem}_fog{beta}{p.suffix}" cv2.imwrite(str(output_dir / fcls / fname), foggy) if __name__ == "__main__": print("[FogGuard] Preparing NMDC mine fog dataset...") DATASET_DIR.mkdir(parents=True, exist_ok=True) prepare_dataset("./data/raw", "./data/processed") print("✅ Dataset preparation complete!")
# train_fog_model.py – MineSafe Model Training # EfficientNet-B0 fog classifier + YOLOv8 obstacle detector import torch, torchvision from torchvision import transforms, datasets from torch.utils.data import DataLoader from ultralytics import YOLO import timm, numpy as np from pathlib import Path # ── Config ───────────────────────────────────────────────── DATA_DIR = Path("./data/processed") MODEL_DIR = Path("./models") EPOCHS = 100 BATCH_SIZE = 32 IMG_SIZE = 224 NUM_CLASSES = 4 # CLEAR, MODERATE, DENSE, CRITICAL DEVICE = "cuda" if torch.cuda.is_available() else "cpu" print(f"[FogGuard] Using device: {DEVICE}") # ── Data Transforms ───────────────────────────────────────── train_tf = transforms.Compose([ transforms.Resize((IMG_SIZE, IMG_SIZE)), transforms.RandomHorizontalFlip(), transforms.ColorJitter(brightness=0.3, contrast=0.3, saturation=0.2), transforms.RandomRotation(10), transforms.ToTensor(), transforms.Normalize(mean=[0.485, 0.456, 0.406], std=[0.229, 0.224, 0.225]) ]) val_tf = transforms.Compose([ transforms.Resize((IMG_SIZE, IMG_SIZE)), transforms.ToTensor(), transforms.Normalize(mean=[0.485, 0.456, 0.406], std=[0.229, 0.224, 0.225]) ]) # ── Dataset ───────────────────────────────────────────────── full_ds = datasets.ImageFolder(DATA_DIR, transform=train_tf) n = len(full_ds) n_train = int(n * 0.8) n_val = int(n * 0.1) n_test = n - n_train - n_val train_ds, val_ds, test_ds = torch.utils.data.random_split( full_ds, [n_train, n_val, n_test]) train_loader = DataLoader(train_ds, batch_size=BATCH_SIZE, shuffle=True, num_workers=4) val_loader = DataLoader(val_ds, batch_size=BATCH_SIZE, shuffle=False, num_workers=4) # ── Model: EfficientNet-B0 (Transfer Learning) ────────────── model = timm.create_model("efficientnet_b0", pretrained=True, num_classes=NUM_CLASSES) model = model.to(DEVICE) criterion = torch.nn.CrossEntropyLoss(label_smoothing=0.1) optimizer = torch.optim.AdamW(model.parameters(), lr=3e-4, weight_decay=1e-4) scheduler = torch.optim.lr_scheduler.CosineAnnealingLR(optimizer, T_max=EPOCHS) # ── Training Loop ──────────────────────────────────────────── best_val_acc = 0.0 for epoch in range(EPOCHS): model.train() total_loss, correct = 0, 0 for imgs, labels in train_loader: imgs, labels = imgs.to(DEVICE), labels.to(DEVICE) optimizer.zero_grad() out = model(imgs) loss = criterion(out, labels) loss.backward() optimizer.step() total_loss += loss.item() correct += (out.argmax(1) == labels).sum().item() train_acc = correct / len(train_ds) # Validation model.eval() val_correct = 0 with torch.no_grad(): for imgs, labels in val_loader: imgs, labels = imgs.to(DEVICE), labels.to(DEVICE) val_correct += (model(imgs).argmax(1) == labels).sum().item() val_acc = val_correct / len(val_ds) print(f"Epoch {epoch+1}/{EPOCHS} | Loss: {total_loss:.3f} | Train: {train_acc:.3f} | Val: {val_acc:.3f}") if val_acc > best_val_acc: best_val_acc = val_acc torch.save(model.state_dict(), MODEL_DIR / "fogguard_best.pt") scheduler.step() print(f"✅ Training complete! Best Val Acc: {best_val_acc:.4f}") # ── YOLOv8 Obstacle Detection ──────────────────────────────── print("[FogGuard] Training YOLOv8 for foggy obstacle detection...") yolo = YOLO("yolov8n.pt") yolo.train( data="./data/yolo_dataset.yaml", epochs=80, imgsz=640, batch=16, name="fogguard_yolo", optimizer="AdamW", lr0=1e-3, augment=True, mosaic=1.0, degrees=10, flipud=0.5 ) print("✅ YOLOv8 training complete!")
# export_model.py – Export to TF.js for browser inference import torch, timm, tensorflowjs as tfjs import tensorflow as tf, numpy as np from pathlib import Path MODEL_PATH = "./models/fogguard_best.pt" EXPORT_DIR = "./model" # → website/model/ directory NUM_CLASSES = 4 # ── Step 1: Load PyTorch model ─────────────────────────────── pt_model = timm.create_model("efficientnet_b0", pretrained=False, num_classes=NUM_CLASSES) pt_model.load_state_dict(torch.load(MODEL_PATH, map_location="cpu")) pt_model.eval() # ── Step 2: Export to ONNX ─────────────────────────────────── dummy = torch.randn(1, 3, 224, 224) torch.onnx.export( pt_model, dummy, "./models/fogguard.onnx", input_names=["input"], output_names=["output"], dynamic_axes={"input": {0: "batch"}, "output": {0: "batch"}}, opset_version=12 ) print("✅ ONNX export complete") # ── Step 3: Convert ONNX → TF SavedModel ──────────────────── # pip install onnx2tf import subprocess subprocess.run(["onnx2tf", "-i", "./models/fogguard.onnx", "-o", "./models/tf_model"]) print("✅ TF SavedModel conversion complete") # ── Step 4: Convert to TF.js ───────────────────────────────── tfjs.converters.save_keras_model( tf.saved_model.load("./models/tf_model"), EXPORT_DIR ) print(f"✅ TF.js model saved to {EXPORT_DIR}") print(" → Copy model.json + group1-shard*.bin to website/model/") # ── Quick Inference Test ───────────────────────────────────── import cv2 def predict_fog(image_path): img = cv2.imread(image_path) img = cv2.resize(img, (224, 224)) / 255.0 # Normalize mean = np.array([0.485, 0.456, 0.406]) std = np.array([0.229, 0.224, 0.225]) img = (img[..., ::-1] - mean) / std # BGR→RGB tensor = torch.tensor(img.transpose(2,0,1), dtype=torch.float32).unsqueeze(0) with torch.no_grad(): logits = pt_model(tensor) probs = torch.softmax(logits, dim=1).squeeze().numpy() classes = ["CLEAR", "MODERATE", "DENSE", "CRITICAL"] pred_cls = classes[np.argmax(probs)] conf = probs.max() print(f"Prediction: {pred_cls} ({conf:.2%} confidence)") return pred_cls, conf # Test on sample image predict_fog("./data/sample_foggy_road.jpg")
